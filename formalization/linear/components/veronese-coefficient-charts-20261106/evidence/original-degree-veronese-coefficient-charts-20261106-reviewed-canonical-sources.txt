module
public import Linear.OriginalDegreeVeroneseAlgebra

noncomputable section
set_option autoImplicit false
set_option maxHeartbeats 5000
namespace LinearStudy
universe u v w
variable {K : Type u} {S : Type v} {M : Type w}
variable [Field K] [CommRing S] [Algebra K S]
variable [AddCommGroup M] [Module K M] [Module S M]
variable (B : ℕ → Submodule K S) [GradedAlgebra B]
variable (D : ℤ → Submodule K M)
variable (hD : ∀ (n : ℕ) (d : ℤ) (c : S), c ∈ B n →
  ∀ x : M, x ∈ D d → c • x ∈ D ((n : ℤ) + d))

/-- The original degree-qn coefficient pieces form a module over the
degree-q Veronese subalgebra. -/
def originalDegreeVeroneseCoefficient (q : ℕ) :
    Submodule (originalDegreeVeroneseAlgebra B q) M := by
  let W : Submodule K M := ⨆ j : ℤ, D ((q : ℤ) * j)
  refine {carrier := W, zero_mem' := W.zero_mem, add_mem' := W.add_mem, smul_mem' := ?_}
  intro c x hx
  have hc : (c : S) ∈ ⨆ n, B (q * n) := c.property
  have hxW : x ∈ W := hx
  change (c : S) • x ∈ W
  refine Submodule.iSup_induction (fun n => B (q * n))
    (motive := fun a => a • x ∈ W) hc ?_ ?_ ?_
  · intro n a ha
    refine Submodule.iSup_induction (fun j : ℤ => D ((q : ℤ) * j))
      (motive := fun y => a • y ∈ W) hxW ?_ ?_ ?_
    · intro j y hy
      apply Submodule.mem_iSup_of_mem ((n : ℤ) + j)
      change a • y ∈ D ((q : ℤ) * ((n : ℤ) + j))
      rw [mul_add, ← Nat.cast_mul]
      exact hD (q * n) ((q : ℤ) * j) a ha y hy
    · simpa only [smul_zero] using W.zero_mem
    · intro y z hy hz
      rw [smul_add]
      exact W.add_mem hy hz
  · simpa only [zero_smul] using W.zero_mem
  · intro a b ha hb
    rw [add_smul]
    exact W.add_mem ha hb

theorem originalDegreeVeroneseCoefficient_mem_of_mem
    (q : ℕ) (j : ℤ) {x : M} (hx : x ∈ D ((q : ℤ) * j)) :
    x ∈ originalDegreeVeroneseCoefficient B D hD q :=
  Submodule.mem_iSup_of_mem j hx

end LinearStudy

variable (q : ℕ)
local notation "V" => originalDegreeVeroneseAlgebra B q
local notation "W" => originalDegreeVeroneseCoefficient B D hD q

/-- Regraded integer piece, pulled back along the actual coefficient inclusion. -/
def originalDegreeVeroneseCoefficientPiece (j : ℤ) : Submodule K W :=
  (D ((q : ℤ) * j)).comap ((W).restrictScalars K).subtype

theorem originalDegreeVeroneseCoefficientPiece_mem_iff
    (j : ℤ) (x : W) :
    x ∈ originalDegreeVeroneseCoefficientPiece B D hD q j ↔
      (x : M) ∈ D ((q : ℤ) * j) := Iff.rfl

/-- The actual Veronese scalar action adds the regraded degrees. -/
theorem originalDegreeVeroneseCoefficient_smul_homogeneous
    (n : ℕ) (j : ℤ) (c : V)
    (hc : c ∈ originalDegreeVeronesePiece B q n)
    (x : W) (hx : x ∈ originalDegreeVeroneseCoefficientPiece B D hD q j) :
    c • x ∈ originalDegreeVeroneseCoefficientPiece B D hD q ((n : ℤ) + j) := by
  have hcS : (c : S) ∈ B (q * n) := hc
  have hxM : (x : M) ∈ D ((q : ℤ) * j) := hx
  change (c : S) • (x : M) ∈ D ((q : ℤ) * ((n : ℤ) + j))
  rw [mul_add, ← Nat.cast_mul]
  exact hD (q * n) ((q : ℤ) * j) (c : S) hcS (x : M) hxM

end LinearStudy

/-- The numerator degree on a regraded chart with an integer shift
matches its original degree exactly. -/
theorem originalDegreeVeroneseChartNumeratorDegree (q d n : ℕ) (k : ℤ) :
    (q : ℤ) * ((n : ℤ) * (d : ℤ) + k) =
      ((n * (q * d) : ℕ) : ℤ) + (q : ℤ) * k := by
  rw [mul_add, Nat.cast_mul, Nat.cast_mul, mul_left_comm (q : ℤ) (n : ℤ)]

    a ∈ originalDegreeVeroneseCoefficientPiece B D hD q ((n : ℤ) * (d : ℤ) + k) ↔
      (a : M) ∈ D (((n * (q * d) : ℕ) : ℤ) + (q : ℤ) * k) := by
  rw [originalDegreeVeroneseCoefficientPiece_mem_iff, originalDegreeVeroneseChartNumeratorDegree]

/-- Every numerator of the original chart degree comes from the regraded
coefficient with the same underlying value. -/
theorem originalDegreeVeroneseChartNumerator_exists
    (d n : ℕ) (k : ℤ) (a : M)
    (ha : a ∈ D (((n * (q * d) : ℕ) : ℤ) + (q : ℤ) * k)) :
    ∃ b : W, (b : M) = a ∧
      b ∈ originalDegreeVeroneseCoefficientPiece B D hD q ((n : ℤ) * (d : ℤ) + k) := by
  have ha' : a ∈ D ((q : ℤ) * ((n : ℤ) * (d : ℤ) + k)) :=
    (originalDegreeVeroneseChartNumeratorDegree q d n k).symm ▸ ha
  let b : W := ⟨a, originalDegreeVeroneseCoefficient_mem_of_mem B D hD q _ ha'⟩
  exact ⟨b, rfl, ha'⟩

/-- The powers used as chart denominators have the same values in S. -/
def originalDegreeVeroneseCoefficientDenominator (f : V) (s : Submonoid.powers f) :
    Submonoid.powers (f : S) :=
  ⟨((s : V) : S), by
    obtain ⟨n, hn⟩ := s.property
    exact ⟨n, congrArg (fun a : V => (a : S)) hn⟩⟩

omit [AddCommGroup M] [Module K M] [Module S M] in
theorem originalDegreeVeroneseCoefficientDenominator_surjective (f : V) :
    Function.Surjective (originalDegreeVeroneseCoefficientDenominator B q f) := by
  intro t
  obtain ⟨n, hn⟩ := t.property
  refine ⟨⟨f ^ n, ⟨n, rfl⟩⟩, Subtype.ext ?_⟩
  exact hn

/-- Send a coefficient fraction to its identical value in the original
localized coefficient module. -/
def originalDegreeVeroneseCoefficientLocalization (f : V)
    (z : LocalizedModule (Submonoid.powers f) W) :
    LocalizedModule (Submonoid.powers (f : S)) M :=
  LocalizedModule.liftOn z
    (fun p => LocalizedModule.mk (p.1 : M)
      (originalDegreeVeroneseCoefficientDenominator B q f p.2))
    (by
      intro p p' hp
      change LocalizedModule.r (Submonoid.powers f) W p p' at hp
      obtain ⟨u, hu⟩ := hp
      exact LocalizedModule.mk_eq.mpr
        ⟨originalDegreeVeroneseCoefficientDenominator B q f u,
          congrArg (fun x : W => (x : M)) hu⟩)

theorem originalDegreeVeroneseCoefficientLocalization_mk
    (f : V) (a : W) (s : Submonoid.powers f) :
    originalDegreeVeroneseCoefficientLocalization B D hD q f (LocalizedModule.mk a s) =
      LocalizedModule.mk (a : M) (originalDegreeVeroneseCoefficientDenominator B q f s) := rfl

theorem originalDegreeVeroneseCoefficientLocalization_injective (f : V) :
    Function.Injective (originalDegreeVeroneseCoefficientLocalization B D hD q f) := by
  intro x y
  refine LocalizedModule.induction_on₂ ?_ x y
  intro a a' s s' h
  rw [originalDegreeVeroneseCoefficientLocalization_mk,
      originalDegreeVeroneseCoefficientLocalization_mk] at h
  obtain ⟨t, ht⟩ := LocalizedModule.mk_eq.mp h
  obtain ⟨u, rfl⟩ := originalDegreeVeroneseCoefficientDenominator_surjective B q f t
  exact LocalizedModule.mk_eq.mpr ⟨u, Subtype.ext ht⟩

end LinearStudy

theorem localizedFractionMap_add
    {R T X Y : Type*} [CommSemiring R] [CommSemiring T]
    [AddCommMonoid X] [AddCommMonoid Y] [Module R X] [Module T Y]
    {U : Submonoid R} {U' : Submonoid T}
    (φ : LocalizedModule U X → LocalizedModule U' Y)
    (i : X →+ Y) (δ : U → U')
    (hδ : ∀ s t, δ (s * t) = δ s * δ t)
    (hsmul : ∀ (s : U) (a : X), i (s • a) = δ s • i a)
    (hmk : ∀ a s, φ (LocalizedModule.mk a s) = LocalizedModule.mk (i a) (δ s))
    (x y : LocalizedModule U X) : φ (x + y) = φ x + φ y := by
  induction x, y using LocalizedModule.induction_on₂
  simp only [LocalizedModule.mk_add_mk, hmk, map_add, hsmul, hδ]

/-- The original inclusion is additive before localizing. -/
def originalDegreeVeroneseCoefficientInclAdd : W →+ M where
  toFun := fun x => x.val
  map_zero' := rfl
  map_add' := fun _ _ => rfl

theorem originalDegreeVeroneseCoefficientInclAdd_denominator_smul (f : V)
    (s : Submonoid.powers f) (a : W) :
    originalDegreeVeroneseCoefficientInclAdd B D hD q (s • a) =
      originalDegreeVeroneseCoefficientDenominator B q f s •
        originalDegreeVeroneseCoefficientInclAdd B D hD q a := rfl

/-- The specialized fraction-additivity proof. -/
def originalDegreeVeroneseCoefficientLocalization_add (f : V) :=
  @localizedFractionMap_add V S W M
    (inferInstanceAs (CommSemiring V)) (inferInstanceAs (CommSemiring S))
    (inferInstanceAs (AddCommMonoid W)) (inferInstanceAs (AddCommMonoid M))
    (Submodule.module W) (inferInstanceAs (Module S M))
    (Submonoid.powers f) (Submonoid.powers (f : S))
    (originalDegreeVeroneseCoefficientLocalization B D hD q f)
    (originalDegreeVeroneseCoefficientInclAdd B D hD q)
    (originalDegreeVeroneseCoefficientDenominator B q f)
    (originalDegreeVeroneseCoefficientDenominator_mul B q f)
    (originalDegreeVeroneseCoefficientInclAdd_denominator_smul B D hD q f)
    (originalDegreeVeroneseCoefficientLocalization_mk B D hD q f)

end LinearStudy

/-- Compare the native weighted chart carriers by their single homogeneous
fractions, with exact numerator degrees and the same powers of the denominators. -/
theorem nativeGradedWeightedChart_image
    (hD : ∀ (n : ℕ) (j : ℤ) (c : R), c ∈ A n →
      ∀ x : X, x ∈ D j → c • x ∈ D ((n : ℤ) + j))

      ∀ y : Y, y ∈ D' j → c • y ∈ D' ((n : ℤ) + j))
    (d e : ℕ) (a : R) (b : T) (ha : a ∈ A d) (hb : b ∈ A' e)
