15: Ext/H inverse and Yoneda composition followed by an EXPLICIT trace.
16: This constructs an additive pairing;no trace existence,perfection,
17: canonical dualizing identification or geometric Serre duality is assumed proved. -/

20:     (a : M ⟶ W) : schemeModuleActualCohomology X M r →+ R where
21:   toFun h := trace (((schemeModuleGlobalExtActualHAddEquiv X M r).symm h).comp
22:     (Ext.mk₀ a) (add_zero r))
23:   map_zero' := by simp only [map_zero,Ext.zero_comp]
24:   map_add' h k := by simp only [map_add,Ext.add_comp]
25: /-- Actual global-section/Ext pairing induced by SAME explicit trace.
26: This uses the original unit-Hom/global-section equivalence. -/
27: def schemeModuleExtSectionTracePairing
28:     (trace : Ext.{u} (SheafOfModules.unit X.ringCatSheaf) W r →+ R)
29:     (z : Ext.{u} E W r) : Γ(E,⊤) →+ R where
30:   toFun s := trace ((Ext.mk₀ ((schemeModuleUnitHomGlobalSectionsAddEquiv X E).symm s)).comp
31:     z (zero_add r))
32:   map_zero' := by simp only [map_zero,Ext.mk₀_zero,Ext.zero_comp]
33:   map_add' s t := by simp only [map_add,Ext.mk₀_add,Ext.add_comp]
34: /-- Push the SAME Ext class forward by an ACTUAL coefficient morphism. -/
35: def schemeModuleExtClassCoefficientPushforwardAddHom (epsilon : Ext.{u} E M r) :
36:     (M ⟶ W) →+ Ext.{u} E W r where
37:   toFun a := epsilon.comp (Ext.mk₀ a) (add_zero r)
38:   map_zero' := by simp only [Ext.mk₀_zero,Ext.comp_zero]
39:   map_add' a b := by simp only [Ext.mk₀_add,Ext.comp_add]
40: /-- Transpose FORMULA for actual section boundary,derived from Yoneda
41: associativity rather than inserted as a compatibility hypothesis.
42: The trace remains EXPLICIT;its geometric existence/perfection is separate. -/
43: theorem schemeModuleExtClassActualSectionBoundary_trace_transpose
44:     (epsilon : Ext.{u} E M r) (a : M ⟶ W)
45:     (trace : Ext.{u} (SheafOfModules.unit X.ringCatSheaf) W r →+ R)
46:     (s : Γ(E,⊤)) :
47:     schemeModuleActualHTracePairing trace a (schemeModuleExtClassActualSectionBoundary epsilon s) =
48:       schemeModuleExtSectionTracePairing trace
49:         (schemeModuleExtClassCoefficientPushforwardAddHom epsilon a) s := by
50:   dsimp only [schemeModuleActualHTracePairing,schemeModuleExtClassActualSectionBoundary,
51:     schemeModuleExtSectionTracePairing,schemeModuleExtClassCoefficientPushforwardAddHom,
52:     AddMonoidHom.coe_mk,ZeroHom.coe_mk]
53:   rw [AddEquiv.symm_apply_apply]
54:   congr 1
55:   exact Ext.comp_assoc_of_third_deg_zero _ _ _ (zero_add r)
56: /-- Same transpose formula for SAME original chain/Koszul Ext roof and
57: SAME actual endpoint morphism. No replacement Ext class is introduced. -/
58: theorem schemeModuleOriginalActualSectionBoundary_trace_transpose
59:     (C : ChainComplex X.Modules ℕ) (r : ℕ)
60:     (alpha : C ⟶ (ChainComplex.single₀ X.Modules).obj E) (hAlpha : QuasiIso alpha)
61:     (hAbove : CategoryTheory.Limits.IsZero (C.X (r+1))) (phi : C.X r ≅ M)
62:     (a : M ⟶ W)
63:     (trace : Ext.{u} (SheafOfModules.unit X.ringCatSheaf) W r →+ R)
64:     (s : Γ(E,⊤)) :
65:     schemeModuleActualHTracePairing trace a
66:         (schemeModuleOriginalActualSectionBoundary C r alpha hAlpha hAbove phi s) =
67:       schemeModuleExtSectionTracePairing trace
68:         (schemeModuleExtClassCoefficientPushforwardAddHom
69:           (schemeModuleOriginalChainTopExtClassSmall C r alpha hAlpha phi.hom
70:             (hAbove.eq_of_src _ _)) a) s := by
71:   exact schemeModuleExtClassActualSectionBoundary_trace_transpose _ a trace s
72: /-- CONDITIONAL consequence of explicit trace NONDEGENERACY and actual
73: boundary surjectivity:coefficient pushforward is injective.
74: No actual geometric perfect pairing,trace or dualizing object is supplied
75: by this theorem;these hypotheses must be discharged independently. -/
76: theorem schemeModuleExtClassCoefficientPushforward_injective_of_trace
77:     (epsilon : Ext.{u} E M r)
78:     (trace : Ext.{u} (SheafOfModules.unit X.ringCatSheaf) W r →+ R)
79:     (hTrace : ∀ a : M ⟶ W,(∀ h : schemeModuleActualCohomology X M r,
80:       schemeModuleActualHTracePairing trace a h = 0) → a = 0)
81:     (hDelta : Function.Surjective (schemeModuleExtClassActualSectionBoundary epsilon)) :
82:     Function.Injective (schemeModuleExtClassCoefficientPushforwardAddHom (W := W) epsilon) := by
83:   intro a b hab
84:   apply sub_eq_zero.mp
85:   apply hTrace (a-b)
86:   intro h
87:   obtain ⟨s,rfl⟩ := hDelta h
88:   rw [schemeModuleExtClassActualSectionBoundary_trace_transpose]
89:   have hz : schemeModuleExtClassCoefficientPushforwardAddHom epsilon (a-b) = 0 := by
90:     rw [map_sub,hab,sub_self]
91:   rw [hz]
92:   simp only [schemeModuleExtSectionTracePairing,Ext.comp_zero,map_zero,AddMonoidHom.coe_mk,ZeroHom.coe_mk]
93: end LinearStudy

===== Linear.FiniteAdditiveDualRepresentative =====

8: /-- A finite dual comparison represents every additive functional linear
9: for the independently specified ORIGINAL scalar operation. Both scalar
10: laws are explicit;the native application proves the comparison law from
11: original coefficient multiplication. No scalar action is transported. -/
12: theorem finiteAdditiveDual_exists_representative
13:     {K H V : Type*} [Field K] [AddCommGroup H] [AddCommGroup V] [Module K V]
14:     [Module.Finite K V] (e : H ≃+ Module.Dual K V) (rho : K → H → H)
15:     (hscalar : ∀ r x,e (rho r x)=r • e x)
16:     (ell : H →+ K) (hlin : ∀ r x,ell (rho r x)=r * ell x) :
17:     ∃ v : V,∀ x,e x v=ell x := by
18:   letI : Module.Free K V := Module.Free.of_divisionRing K V
19:   letI : Module.IsReflexive K V := Module.IsReflexive.of_finite_of_free K V
20:   have hinv (r : K) (y : Module.Dual K V) : e.symm (r • y)=rho r (e.symm y) := by
21:     apply e.injective
22:     rw [e.apply_symm_apply,hscalar,e.apply_symm_apply]
23:   let g : Module.Dual K (Module.Dual K V) := {
24:     toFun := fun y => ell (e.symm y)
25:     map_add' := fun y z => by rw [map_add,map_add]
26:     map_smul' := fun r y => by
27:       change ell (e.symm (r • y))=r * ell (e.symm y)
28:       rw [hinv,hlin] }
29:   refine ⟨(Module.evalEquiv K V).symm g,?_⟩
30:   intro x
31:   rw [Module.apply_evalEquiv_symm_apply]
32:   change ell (e.symm (e x))=ell x
33:   rw [e.symm_apply_apply]
34: end LinearStudy
