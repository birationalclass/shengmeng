  [inst : CommRing R] →
    [inst_1 : CommRing T] →
      {r : ℕ} →
        (e : R ≃+* T) →
          (H : Fin r → R) →
            LinearStudy.homogeneousKoszulCoordinateComplex H ≅
              ((ModuleCat.restrictScalars e.toRingHom).mapHomologicalComplex (ComplexShape.down ℕ)).obj
                (LinearStudy.homogeneousKoszulCoordinateComplex (⇑e ∘ H)) :=
fun {R T : Type u} [CommRing R] [CommRing T] {r : ℕ} (e : R ≃+* T) (H : Fin r → R) =>
  HomologicalComplex.Hom.isoOfComponents
    (fun (p : ℕ) =>
      have E :=
        { toFun := fun (v : ↑(Set.powersetCard (Fin r) p) → R) (i : ↑(Set.powersetCard (Fin r) p)) => e (v i),
          map_add' := ⋯, map_smul' := ⋯,
          invFun :=
            fun (v : ↑((ModuleCat.restrictScalars e.toRingHom).obj ↧(↑(Set.powersetCard (Fin r) p) → T)))
              (i : ↑(Set.powersetCard (Fin r) p)) =>
            e.symm (v i),
          left_inv := ⋯, right_inv := ⋯ };
      E.toModuleIso)
    ⋯
