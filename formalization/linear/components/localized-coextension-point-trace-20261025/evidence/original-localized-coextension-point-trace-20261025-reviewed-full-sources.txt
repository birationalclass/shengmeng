/-- Weak regularity of the ACTUAL ring sequence is detected at ALL actual
maximal localizations. No global CM,finiteness or freeness is an input. -/
theorem weaklyRegular_of_maximal_localizations
    {A : Type*} [CommRing A] (rs : List A)
    (hlocal : ∀ (P : Ideal A) [P.IsMaximal],
      IsWeaklyRegular (Localization.AtPrime P)
        (rs.map (algebraMap A (Localization.AtPrime P)))) :
    IsWeaklyRegular A rs := by
  refine ⟨fun i hi => ?_⟩
  let I := Ideal.ofList (rs.take i)
  rw [ideal_smul_top_self]
  apply (isSMulRegular_quotient_iff_mem_of_smul_mem I rs[i]).mpr
  intro a ha
  apply Ideal.mem_of_localization_maximal
  intro P hP
  let L := Localization.AtPrime P
  let f : A →+* L := algebraMap A L
  have hl := (hlocal P).regular_mod_prev i (by simpa using hi)
  change IsSMulRegular (L ⧸ (Ideal.ofList ((rs.map f).take i) •
    (⊤ : Submodule L L))) ((rs.map f)[i]'(by simpa using hi)) at hl
  rw [← List.map_take,← Ideal.map_ofList,ideal_smul_top_self] at hl
  simp only [List.getElem_map] at hl
  apply (isSMulRegular_quotient_iff_mem_of_smul_mem (I.map f) (f rs[i])).mp hl
  change f rs[i] * f a ∈ I.map f
  rw [← map_mul]
  exact Ideal.mem_map_of_mem f ha
/-- Derive finite freeness from ACTUAL source-point regularity and the
base local maximal parameters,using the existing regular-generators theorem.
This is a local algebra criterion only;original geometric local-ring
identifications and regularity inputs are NOT asserted or weakened. -/
theorem finite_free_of_maximal_local_regular_parameters
    {R S : Type*} [CommRing R] [IsLocalRing R] [IsNoetherianRing R]
    [CommRing S] [Algebra R S] [Module.Finite R S]
    (rs : List R) (hgen : Ideal.ofList rs=IsLocalRing.maximalIdeal R)
    (hlocal : ∀ (P : Ideal S) [P.IsMaximal],
      IsWeaklyRegular (Localization.AtPrime P)
        ((rs.map (algebraMap R S)).map (algebraMap S (Localization.AtPrime P)))) :
    Module.Free R S :=
  free_of_regular_maximal_algebraMap rs
    (weaklyRegular_of_maximal_localizations _ hlocal) hgen
end LinearStudy

/-- Finiteness alone derives Artinianity of the ACTUAL closed fiber of a
finite algebra over a local ring. No reducedness or flatness is assumed. -/
theorem finite_algebra_closed_fiber_artinian
    {R S : Type*} [CommRing R] [IsLocalRing R] [CommRing S]
    [Algebra R S] [Module.Finite R S] :
    IsArtinianRing (S ⧸ (IsLocalRing.maximalIdeal R).map (algebraMap R S)) := by
  let m := IsLocalRing.maximalIdeal R
  let I := m.map (algebraMap R S)
  letI : Field (R ⧸ m) := Ideal.Quotient.field m
  letI : Algebra (R ⧸ m) (S ⧸ I) :=
    Ideal.Quotient.algebraQuotientOfLEComap (R:=R) (A:=S) (p:=m) (P:=I) Ideal.le_comap_map
  letI : IsScalarTower R (R ⧸ m) (S ⧸ I) :=
    IsScalarTower.of_algebraMap_eq (fun _ => rfl)
  letI : Module.Finite R (S ⧸ I) := Module.Finite.of_surjective
    (Ideal.Quotient.mkₐ R I).toLinearMap Ideal.Quotient.mk_surjective
  letI : Module.Finite (R ⧸ m) (S ⧸ I) :=
    Module.Finite.of_restrictScalars_finite R (R ⧸ m) (S ⧸ I)
  exact IsArtinianRing.of_finite (R ⧸ m) (S ⧸ I)
/-- Every actual source maximal ideal of a finite local-base algebra lies
above the base maximal ideal. The fiber-support inclusion is derived. -/
theorem finite_algebra_maximal_contains_base_maximal
    {R S : Type*} [CommRing R] [IsLocalRing R] [CommRing S]
    [Algebra R S] [Module.Finite R S] (P : Ideal S) [P.IsMaximal] :
    (IsLocalRing.maximalIdeal R).map (algebraMap R S) ≤ P := by
  have hc := Ideal.isMaximal_under_of_isIntegral_of_isMaximal (R:=R) (S:=S) P
  have he : P.comap (algebraMap R S)=IsLocalRing.maximalIdeal R :=
    IsLocalRing.eq_maximalIdeal hc
  exact Ideal.map_le_iff_le_comap.mpr he.symm.le
/-- In the ACTUAL reduced closed fiber,base maximal parameters generate
each actual source maximal ideal after localization. LOCAL CM and local
dimension are explicit here;no global CM on the original variety is added. -/
theorem finite_reduced_fiber_parameters_regular_at_maximal
    {R S : Type*} [CommRing R] [IsLocalRing R] [IsNoetherianRing R]
    [CommRing S] [Algebra R S] [Module.Finite R S]
    [IsReduced (S ⧸ (IsLocalRing.maximalIdeal R).map (algebraMap R S))]
    (rs : List R) (hgen : Ideal.ofList rs=IsLocalRing.maximalIdeal R)
    (P : Ideal S) [P.IsMaximal]
    [IsCohenMacaulayLocalRing (Localization.AtPrime P)]
    (hlen : (rs.length : WithBot ℕ∞)=ringKrullDim (Localization.AtPrime P)) :
    IsRegular (Localization.AtPrime P)
      ((rs.map (algebraMap R S)).map (algebraMap S (Localization.AtPrime P))) := by
  letI : IsNoetherianRing S := IsNoetherianRing.of_finite R S
  let I := Ideal.ofList (rs.map (algebraMap R S))
  have hI : I=(IsLocalRing.maximalIdeal R).map (algebraMap R S) := by
    dsimp only [I]
    rw [← Ideal.map_ofList,hgen]
  letI : IsArtinianRing (S ⧸ I) := by
    rw [hI]
    exact finite_algebra_closed_fiber_artinian (R:=R) (S:=S)
  letI : IsReduced (S ⧸ I) := by rw [hI];infer_instance
  have hIP : I ≤ P := hI.le.trans (finite_algebra_maximal_contains_base_maximal P)
  have hloc : Ideal.ofList
      ((rs.map (algebraMap R S)).map (algebraMap S (Localization.AtPrime P)))=
        IsLocalRing.maximalIdeal (Localization.AtPrime P) := by
    rw [← Ideal.map_ofList]
    exact artinian_reduced_cut_local_ideal_eq_maximalIdeal_of_le I P hIP
  apply isRegular_of_maximalIdeal_mem_ofList_minimalPrimes
  · rw [hloc,Ideal.minimalPrimes_eq_subsingleton_self]
    exact Set.mem_singleton _
  · simpa only [List.length_map] using hlen
/-- A LOCAL finite reduced fiber and pointwise LOCAL CM/dimension derive
freeness;the free/flat conclusion is not an input. Instantiation of the
actual original geometric charts remains a separate obligation. -/
theorem finite_free_of_reduced_fiber_local_parameters
    {R S : Type*} [CommRing R] [IsLocalRing R] [IsNoetherianRing R]
    [CommRing S] [Algebra R S] [Module.Finite R S]
    [IsReduced (S ⧸ (IsLocalRing.maximalIdeal R).map (algebraMap R S))]
    [∀ (P : Ideal S) [P.IsMaximal],IsCohenMacaulayLocalRing (Localization.AtPrime P)]
    (rs : List R) (hgen : Ideal.ofList rs=IsLocalRing.maximalIdeal R)
    (hlen : ∀ (P : Ideal S) [P.IsMaximal],
      (rs.length : WithBot ℕ∞)=ringKrullDim (Localization.AtPrime P)) :
    Module.Free R S := by
  apply finite_free_of_maximal_local_regular_parameters rs hgen
  intro P hP
  exact (finite_reduced_fiber_parameters_regular_at_maximal rs hgen P (hlen P)).1
end LinearStudy

/-- ACTUAL module localization followed by the SAME residue base change
is the original module fiber. No freeness or flatness hypothesis. -/
def originalLocalizedModuleFiberEquiv :
    K ⊗[Localization P] LocalizedModule P M ≃ₗ[K] K ⊗[R] M :=
  (TensorProduct.AlgebraTensorModule.congr (LinearEquiv.refl K K)

      a ⊗ₜ[R] m := by
  simp [originalLocalizedModuleFiberEquiv,LocalizedModule.equivTensorProduct_apply_mk,
    Localization.mk_one_eq_algebraMap]

    (originalLocalizedModuleFiberEquiv P M).symm (a ⊗ₜ[R] m)=
      a ⊗ₜ[Localization P] LocalizedModule.mk m 1 := by
  apply (originalLocalizedModuleFiberEquiv P M).injective
  rw [LinearEquiv.apply_symm_apply,originalLocalizedModuleFiberEquiv_tmul]

/-- ACTUAL module fiber comparison for an arbitrary actual base
localization,with mathlib original abstract-localization scalar action.
No action is defined by transport through this comparison. -/
def abstractLocalizedModuleFiberEquiv :
    letI : Module Q (LocalizedModule P M) := LocalizedModule.moduleOfIsLocalization
    K ⊗[Q] LocalizedModule P M ≃ₗ[K] K ⊗[R] M := by
  letI : Module Q (LocalizedModule P M) := LocalizedModule.moduleOfIsLocalization
  let e : Q ⊗[R] M ≃ₗ[Q] LocalizedModule P M :=
    (IsLocalizedModule.isBaseChange P Q (LocalizedModule.mkLinearMap P M)).equiv
  exact (TensorProduct.AlgebraTensorModule.congr (LinearEquiv.refl K K) e.symm).trans
    (TensorProduct.AlgebraTensorModule.cancelBaseChange R Q K K M)
/-- ORIGINAL numerator tensors are preserved by the actual comparison. -/
theorem abstractLocalizedModuleFiberEquiv_tmul (a : K) (m : M) :
    letI : Module Q (LocalizedModule P M) := LocalizedModule.moduleOfIsLocalization
    abstractLocalizedModuleFiberEquiv (Q:=Q) P M (a ⊗ₜ[Q] LocalizedModule.mk m 1)=
      a ⊗ₜ[R] m := by
  letI : Module Q (LocalizedModule P M) := LocalizedModule.moduleOfIsLocalization
  let e : Q ⊗[R] M ≃ₗ[Q] LocalizedModule P M :=
    (IsLocalizedModule.isBaseChange P Q (LocalizedModule.mkLinearMap P M)).equiv
  have he : e ((1:Q) ⊗ₜ[R] m)=LocalizedModule.mk m 1 := by
    simp [e,IsBaseChange.equiv_tmul]
  have hes : e.symm (LocalizedModule.mk m 1)=((1:Q) ⊗ₜ[R] m) :=
    e.symm_apply_eq.mpr he.symm
  change TensorProduct.AlgebraTensorModule.cancelBaseChange R Q K K M
    ((TensorProduct.AlgebraTensorModule.congr (LinearEquiv.refl K K) e.symm)
      (a ⊗ₜ[Q] LocalizedModule.mk m 1))=_
  rw [TensorProduct.AlgebraTensorModule.congr_tmul,LinearEquiv.refl_apply,hes,
    TensorProduct.AlgebraTensorModule.cancelBaseChange_tmul,one_smul]
/-- The inverse uses the SAME original numerator. -/
theorem abstractLocalizedModuleFiberEquiv_symm_tmul (a : K) (m : M) :
    letI : Module Q (LocalizedModule P M) := LocalizedModule.moduleOfIsLocalization
    (abstractLocalizedModuleFiberEquiv (Q:=Q) P M).symm (a ⊗ₜ[R] m)=
      a ⊗ₜ[Q] LocalizedModule.mk m 1 := by
  letI : Module Q (LocalizedModule P M) := LocalizedModule.moduleOfIsLocalization
  apply (abstractLocalizedModuleFiberEquiv (Q:=Q) P M).injective
  rw [LinearEquiv.apply_symm_apply,abstractLocalizedModuleFiberEquiv_tmul]

theorem localizedCoextensionOriginalFiberTrace_actual_point_exists_one
    (s : S) (he : ((1:K) ⊗ₜ[R] s : (K ⊗[R] S)) ≠ 0)

      coextensionOriginalBaseChangedEvaluationAtOne m=1 := by
  let e : (K ⊗[R] S) := (1:K) ⊗ₜ[R] s
  obtain ⟨eta,heta,htrace⟩ := coextensionFieldTrace_idempotent_fixed_exists_one
    (K:=K) e he hi
  let f : Module.Dual K (K ⊗[R] S) :=
    coextensionOriginalDualEquiv (R:=K) (S:=(K ⊗[R] S)) eta
  let m : (K ⊗[R] ((ModuleCat.restrictScalars (algebraMap R S)).obj ((ModuleCat.coextendScalars (algebraMap R S)).obj (ModuleCat.of R R)))) := (localizedCoextensionOriginalDualTensorEquiv (R:=R) (S:=S) (Q:=Q) (K:=K) P hfree hinj).symm f
  have hm : localizedCoextensionOriginalDualTensorEquiv P hfree hinj m=f :=
    (localizedCoextensionOriginalDualTensorEquiv (R:=R) (S:=S) (Q:=Q) (K:=K) P hfree hinj).apply_symm_apply f
  refine ⟨m,?_,?_⟩
  · apply (localizedCoextensionOriginalDualTensorEquiv (R:=R) (S:=S) (Q:=Q) (K:=K) P hfree hinj).injective
    apply LinearMap.ext
    intro z
    calc
      _ = localizedCoextensionOriginalDualTensorEquiv P hfree hinj m (z*e) :=
        localizedCoextensionOriginalDualTensorEquiv_scalar_naturality P hfree hinj s m z
      _ = f (z*e) := LinearMap.congr_fun hm _
      _ = (coextensionOriginalDualEquiv (R:=K) (S:=(K ⊗[R] S)) (e • eta)) z := by
        change eta (z*e)=(e • eta) z
        rw [ModuleCat.CoextendScalars.smul_apply]
      _ = f z := by rw [heta]
      _ = _ := (LinearMap.congr_fun hm z).symm
  · rw [← localizedCoextensionOriginalDualTensorEquiv_evaluation P hfree hinj]
    rw [hm]
    exact htrace

theorem localizedCoextensionOriginalScalarFiberEnd_eq_of_fiber_eq
    (s t : S) (h : ((1:K) ⊗ₜ[R] s : (K ⊗[R] S))=((1:K) ⊗ₜ[R] t)) :

theorem localizedCoextensionOriginalResidueFiberPointEnd_of_lift
    (hres : Function.Surjective (algebraMap R K)) (e : (K ⊗[R] S))
    (s : S) (hs : ((1:K) ⊗ₜ[R] s : (K ⊗[R] S))=e) :
    coextensionOriginalResidueFiberPointEnd hres e=
      coextensionOriginalScalarFiberEnd (R:=R) (K:=K) s := by
  unfold coextensionOriginalResidueFiberPointEnd
  apply localizedCoextensionOriginalScalarFiberEnd_eq_of_fiber_eq P hfree hinj
  have hc := Classical.choose_spec (Algebra.TensorProduct.includeRight_surjective
    (R:=R) (S:=K) (T:=S) hres e)

      (Algebra.TensorProduct.includeRight_surjective (R:=R) (S:=K) (T:=S) hres e) : (K ⊗[R] S))=e := by
    simpa only [Algebra.TensorProduct.includeRight_apply] using hc
  exact hc'.trans hs.symm

theorem localizedCoextensionOriginalResidueFiberPointEnd_idempotent
    (hres : Function.Surjective (algebraMap R K)) (e : (K ⊗[R] S)) (hi : e*e=e) :
    ∀ m,coextensionOriginalResidueFiberPointEnd hres e

    simpa only [Algebra.TensorProduct.includeRight_apply] using hs

theorem localizedCoextensionOriginalResidueFiberPointTrace_exists_one
    (hres : Function.Surjective (algebraMap R K)) (e : (K ⊗[R] S))
    (he : e ≠ 0) (hi : e*e=e) :
    ∃ m : (K ⊗[R] ((ModuleCat.restrictScalars (algebraMap R S)).obj ((ModuleCat.coextendScalars (algebraMap R S)).obj (ModuleCat.of R R)))),coextensionOriginalResidueFiberPointEnd hres e m=m ∧
