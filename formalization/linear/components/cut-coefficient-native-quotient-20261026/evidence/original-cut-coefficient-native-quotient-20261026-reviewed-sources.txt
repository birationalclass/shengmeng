/-- Actual ORIGINAL coefficient quotient by the extended fiber ideal is
its actual residue tensor. No flatness,finiteness or arbitrary iso input.
This construction will be applied to the actual coextension coefficient. -/
def surjectiveBaseCoefficientQuotientEquiv
    (hres : Function.Surjective (algebraMap R K)) :
    (N ⧸ ((RingHom.ker (algebraMap R K)).map (algebraMap R S) •
      (⊤ : Submodule S N))) ≃ₗ[R] K ⊗[R] N := by
  let m := RingHom.ker (algebraMap R K)
  let P := m.map (algebraMap R S) • (⊤ : Submodule S N)
  have hP : P.restrictScalars R=m • (⊤ : Submodule R N) := by
    simpa only [P,Submodule.restrictScalars_top] using
      (Submodule.restrictScalars_map_smul_eq m (⊤ : Submodule S N))
  let e : (R ⧸ m) ≃ₐ[R] K :=
    Ideal.quotientKerAlgEquivOfSurjective (f:=Algebra.ofId R K) hres
  exact (Submodule.Quotient.restrictScalarsEquiv R P).symm.trans
    ((Submodule.quotEquivOfEq _ _ hP).trans
      ((TensorProduct.quotTensorEquivQuotSMul N m).symm.trans
        (TensorProduct.congr e.toLinearEquiv (LinearEquiv.refl R N))))
/-- The actual quotient comparison retains EVERY original coefficient. -/
theorem surjectiveBaseCoefficientQuotientEquiv_mk
    (hres : Function.Surjective (algebraMap R K)) (n : N) :
    surjectiveBaseCoefficientQuotientEquiv (S:=S) hres (Submodule.Quotient.mk n)=
      ((1:K) ⊗ₜ[R] n) := by
  simp only [surjectiveBaseCoefficientQuotientEquiv,LinearEquiv.trans_apply,
    Submodule.Quotient.restrictScalarsEquiv_symm_mk,Submodule.quotEquivOfEq_mk,
    TensorProduct.quotTensorEquivQuotSMul_symm_mk,TensorProduct.congr_tmul]
  change ((Ideal.quotientKerAlgEquivOfSurjective (f:=Algebra.ofId R K) hres)
    (1 : R ⧸ RingHom.ker (algebraMap R K))) ⊗ₜ[R] n=_
  simp only [map_one]

/-- The SAME original coefficient quotient comparison is linear over the
ORIGINAL coefficient field. This does not transport a source action. -/
def surjectiveBaseCoefficientQuotientOriginalFieldEquiv
    (hres : Function.Surjective (algebraMap R K)) :
    (N ⧸ ((RingHom.ker (algebraMap R K)).map (algebraMap R S) •
      (⊤ : Submodule S N))) ≃ₗ[K] K ⊗[R] N :=
  (surjectiveBaseCoefficientQuotientEquiv (S:=S) hres).restrictScalars K
/-- Original-field linearity leaves every original coefficient class unchanged. -/
theorem surjectiveBaseCoefficientQuotientOriginalFieldEquiv_mk
    (hres : Function.Surjective (algebraMap R K)) (n : N) :
    surjectiveBaseCoefficientQuotientOriginalFieldEquiv (S:=S) hres
      (Submodule.Quotient.mk n)=((1:K) ⊗ₜ[R] n) :=
  surjectiveBaseCoefficientQuotientEquiv_mk (R:=R) (S:=S) (K:=K) (N:=N) hres n

theorem projectivePulledCutEquation_coefficient_quotient_dual_of_fiber_dual {n r : ℕ}
    (f : HomogeneousEndomorphism n) (V : IntegralProjectiveEquations n)
    (L : Fin (r+1) → CoordinateRing n) (w : Fin (r+1) → ℂ)
    (hdual : projectivePulledCutEquationCoefficientFiberDualStatement f V L w) :
    projectivePulledCutEquationCoefficientQuotientDualStatement f V L w := by
  let R := MvPolynomial (Fin r) ℂ
  let B := MvPolynomial (Fin n) ℂ ⧸ V.affineIdeal
  let phi := projectivePulledCutEquationMap f V L w
  let rho : R →ₐ[ℂ] ℂ := MvPolynomial.aeval (0 : Fin r → ℂ)
  letI : Algebra R B := phi.toRingHom.toAlgebra
  letI : SMul R B := phi.toRingHom.toAlgebra.toSMul
  letI : Module R B := @Algebra.toModule R B _ _ phi.toRingHom.toAlgebra
  letI : Algebra R ℂ := rho.toRingHom.toAlgebra
  letI : IsScalarTower ℂ R B := IsScalarTower.of_algHom phi
  letI : IsScalarTower ℂ R ℂ := IsScalarTower.of_algHom rho
  let D := (ModuleCat.coextendScalars (algebraMap R B)).obj (ModuleCat.of R R)
  letI : IsScalarTower ℂ B D := IsScalarTower.of_compHom ℂ B D
  have hres : Function.Surjective (algebraMap R ℂ) :=
    fun a => ⟨MvPolynomial.C a,rho.commutes a⟩
  let I := Ideal.span (Set.range (fun i : Fin r => Ideal.Quotient.mk V.affineIdeal
    (affineChartPolynomialMap (MvPolynomial.aeval f.forms (projectiveLinearSectionForms L w i)))))
  let m := RingHom.ker (algebraMap R ℂ)
  have hI : m.map (algebraMap R B)=I :=
    projectivePulledCutEquationMap_zero_fiber_ideal f V L w
  let Qe := (Submodule.quotEquivOfEq (I • (⊤ : Submodule B D))
    (m.map (algebraMap R B) • (⊤ : Submodule B D))
      (congrArg (fun T : Ideal B => T • (⊤ : Submodule B D)) hI.symm)).restrictScalars ℂ
  let E0 := coextensionOriginalCoefficientQuotientFiberEquiv (K:=ℂ) (R:=R) (S:=B) hres
  unfold projectivePulledCutEquationCoefficientFiberDualStatement at hdual
  dsimp only at hdual
  obtain ⟨E,hpure,hunit,hscalar⟩ := hdual
  unfold projectivePulledCutEquationCoefficientQuotientDualStatement
  dsimp only
  refine ⟨Qe.trans (E0.trans E),?_,?_,?_⟩
  · intro ell P
    change E (E0 (Submodule.Quotient.mk ell)) (Ideal.Quotient.mk _ P)=_
    rw [coextensionOriginalCoefficientQuotientFiberEquiv_mk]
    simpa only [one_mul] using hpure (1:ℂ) ell P
  · intro ell
    change E (E0 (Submodule.Quotient.mk ell)) 1=_
    rw [coextensionOriginalCoefficientQuotientFiberEquiv_mk]
    simpa only [one_mul,map_one] using hpure (1:ℂ) ell (1 : MvPolynomial (Fin n) ℂ)
  · intro P z q
    induction z using Submodule.Quotient.induction_on with
    | _ ell =>
      change E (E0 (Submodule.Quotient.mk ((Ideal.Quotient.mk V.affineIdeal P) • ell))) q=
        E (E0 (Submodule.Quotient.mk ell)) (q*Ideal.Quotient.mk _ P)
      rw [coextensionOriginalCoefficientQuotientFiberEquiv_mk,
        coextensionOriginalCoefficientQuotientFiberEquiv_mk]
      change E (coextensionOriginalScalarFiberEnd (R:=R) (K:=ℂ)
        (Ideal.Quotient.mk V.affineIdeal P) ((1:ℂ) ⊗ₜ[R] ell)) q=_
      exact hscalar P ((1:ℂ) ⊗ₜ[R] ell) q

    (show (ModuleCat.coextendScalars (algebraMap K Q)).obj (ModuleCat.of K K) from

    (L : Fin (r+1) → CoordinateRing n) (hL : ∀ i,(L i).IsHomogeneous 1)

    f V hrn hq hf hV L hL hinj hfinite x0 hx0
  refine ⟨c,hc,hex,?_⟩
  intro w hw
  exact ⟨(hcprop w hw).1,projectivePulledCutEquation_native_dual_of_quotient_dual
    f V L w (hcprop w hw).2⟩
