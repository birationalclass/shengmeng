def CategoryTheory.Functor.mapFormalCoproduct.{w, u, v, u₂, v₂} : {C : Type u} →
  [inst : CategoryTheory.Category.{v, u} C] →
    {D : Type u₂} →
      [inst_1 : CategoryTheory.Category.{v₂, u₂} D] →
        CategoryTheory.Functor C D →
          CategoryTheory.Functor (CategoryTheory.Limits.FormalCoproduct C) (CategoryTheory.Limits.FormalCoproduct D) :=
fun {C : Type u} [CategoryTheory.Category.{v, u} C] {D : Type u₂} [CategoryTheory.Category.{v₂, u₂} D]
    (F : CategoryTheory.Functor C D) =>
  { obj := fun (X : CategoryTheory.Limits.FormalCoproduct C) => { I := X.I, obj := F.obj ∘ X.obj },
    map := fun {X Y : CategoryTheory.Limits.FormalCoproduct C} (f : X ⟶ Y) =>
      { f := f.f, φ := fun (i : { I := X.I, obj := F.obj ∘ X.obj }.I) => F.map (f.φ i) },
    map_id := ⋯, map_comp := ⋯ }
