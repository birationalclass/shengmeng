/-- Actual proof term:transport an actual product restriction square to the
ORIGINAL ordered Cech face. The formal-coproduct face is handled once here,
independently of any graded-ring or native-module implementation. -/
def orderedAbelianOpenSectionComparisonIso_restrict
    {X Y : Scheme.{u}} {ι : Type u} [LinearOrder ι]
    (F : X.Opensᵒᵖ ⥤ AddCommGrpCat.{u}) (G : Y.Opensᵒᵖ ⥤ AddCommGrpCat.{u})
    (U : ι → X.Opens) (V : ι → Y.Opens)
    (p : ℕ) (k : Fin (p+2)) (t : OrderedCechIndex ι (p+1))
    (U0' U1' : X.Opens) (V0' V1' : Y.Opens)
    (hU0 : (∏ᶜ fun j : Fin (p+1) => U ((t.delete k).1 j))=U0')
    (hU1 : (∏ᶜ fun j : Fin (p+2) => U (t.1 j))=U1')
    (hV0 : (∏ᶜ fun j : Fin (p+1) => V ((t.delete k).1 j))=V0')
    (hV1 : (∏ᶜ fun j : Fin (p+2) => V (t.1 j))=V1')
    (fp : U1' ⟶ U0') (gp : V1' ⟶ V0')
    (e0 : F.obj (op U0') ≅ G.obj (op V0')) (e1 : F.obj (op U1') ≅ G.obj (op V1'))
    (he : e0.hom ≫ G.map gp.op = F.map fp.op ≫ e1.hom) :=
  abelianOpenSectionComparisonIso_restrict F G
    (∏ᶜ fun j : Fin (p+1) => U ((t.delete k).1 j)) U0'
    (∏ᶜ fun j : Fin (p+2) => U (t.1 j)) U1'
    (∏ᶜ fun j : Fin (p+1) => V ((t.delete k).1 j)) V0'
    (∏ᶜ fun j : Fin (p+2) => V (t.1 j)) V1'
    hU0 hU1 hV0 hV1 ((orderedCechFace U p k).φ t) fp ((orderedCechFace V p k).φ t) gp e0 e1 he
end LinearStudy

/-- Construct the ORIGINAL ordered Cech complex isomorphism from actual
product-chart isomorphisms and their proved restriction squares. Every
intersection transport,original coface and integer sign is retained. -/
def orderedCrossSpaceAbelianCechComplexIso_of_productCharts
    {X Y : Scheme.{u}} {ι : Type u} [LinearOrder ι]
    (F : X.Opensᵒᵖ ⥤ AddCommGrpCat.{u}) (G : Y.Opensᵒᵖ ⥤ AddCommGrpCat.{u})
    (U : ι → X.Opens) (V : ι → Y.Opens)
    (P : ∀ p : ℕ,OrderedCechIndex ι p → X.Opens)
    (Q : ∀ p : ℕ,OrderedCechIndex ι p → Y.Opens)
    (hP : ∀ p t,(∏ᶜ fun j : Fin (p+1) => U (t.1 j))=P p t)
    (hQ : ∀ p t,(∏ᶜ fun j : Fin (p+1) => V (t.1 j))=Q p t)
    (rP : ∀ (p : ℕ) (k : Fin (p+2)) (t : OrderedCechIndex ι (p+1)),P (p+1) t ⟶ P p (t.delete k))
    (rQ : ∀ (p : ℕ) (k : Fin (p+2)) (t : OrderedCechIndex ι (p+1)),Q (p+1) t ⟶ Q p (t.delete k))
    (e : ∀ p t,F.obj (op (P p t)) ≅ G.obj (op (Q p t)))
    (he : ∀ (p : ℕ) (k : Fin (p+2)) (t : OrderedCechIndex ι (p+1)),
      (e p (t.delete k)).hom ≫ G.map (rQ p k t).op = F.map (rP p k t).op ≫ (e (p+1) t).hom) :=
  orderedCrossSpaceAbelianCechComplexIso F G U V
    (fun p t => abelianOpenSectionComparisonIso F G _ (P p t) _ (Q p t) (hP p t) (hQ p t) (e p t))
    (fun p k t => orderedAbelianOpenSectionComparisonIso_restrict F G U V p k t
      (P p (t.delete k)) (P (p+1) t) (Q p (t.delete k)) (Q (p+1) t)
      (hP p (t.delete k)) (hP (p+1) t) (hQ p (t.delete k)) (hQ (p+1) t)
      (rP p k t) (rQ p k t) (e p (t.delete k)) (e (p+1) t) (he p k t))
end LinearStudy

  dsimp only [nativeCechProductOpen]
  rw [orderedCechIndex_delete_values,nativeCechCoordinateProduct_delete_mul a p t.1 k]

    nativeBaseSourceCechProductOpen B a (p+1) t ⟶ nativeBaseSourceCechProductOpen B a p (t.delete k) := homOfLE (by
  dsimp only [nativeBaseSourceCechProductOpen]
  rw [orderedCechIndex_delete_values,nativeCechCoordinateProduct_delete_mul a p t.1 k,map_mul]
