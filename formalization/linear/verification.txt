def LinearStudy.LinearityTheoremGoal : Prop :=
∀ (n : ℕ) (f : LinearStudy.HomogeneousEndomorphism n) (V : LinearStudy.IntegralProjectiveEquations n),
  1 < f.degree → Function.Surjective f.onPoints → f.onPoints ⁻¹' V.zeroSet = V.zeroSet → ∃ W, ↑W = V.zeroSet
structure LinearStudy.HomogeneousEndomorphism (n : ℕ) : Type
number of parameters: 1
fields:
  LinearStudy.HomogeneousEndomorphism.degree : ℕ
  LinearStudy.HomogeneousEndomorphism.forms : Fin (n + 1) → LinearStudy.CoordinateRing n
  LinearStudy.HomogeneousEndomorphism.homogeneous : ∀ (i : Fin (n + 1)),
      MvPolynomial.IsHomogeneous (self.forms i) self.degree
  LinearStudy.HomogeneousEndomorphism.noBasePoint : ∀ (v : LinearStudy.CoordinateVector n),
      v ≠ 0 → (fun i => (MvPolynomial.eval v) (self.forms i)) ≠ 0
constructor:
  LinearStudy.HomogeneousEndomorphism.mk {n : ℕ} (degree : ℕ) (forms : Fin (n + 1) → LinearStudy.CoordinateRing n)
    (homogeneous : ∀ (i : Fin (n + 1)), MvPolynomial.IsHomogeneous (forms i) degree)
    (noBasePoint : ∀ (v : LinearStudy.CoordinateVector n), v ≠ 0 → (fun i => (MvPolynomial.eval v) (forms i)) ≠ 0) :
    LinearStudy.HomogeneousEndomorphism n
